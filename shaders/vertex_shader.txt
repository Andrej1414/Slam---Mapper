#version 460 core

layout(location = 0) in vec3 PointPos;
layout(location = 1) in vec3 PointColor;
layout(location = 2) in vec3 PosePos;
layout(location = 3) in mat4 InstanceMatrix;

uniform mat4 Perspective;
uniform mat4 View;
uniform bool DrawPoints;

out vec4 VertexColor;

void main(){
    if (DrawPoints){
        gl_Position = Perspective * View * vec4(PointPos,1.0);
        VertexColor = vec4(PointColor,1.0);
    }
    else{
        gl_Position = Perspective * View * InstanceMatrix * vec4(PosePos,1.0);   
        VertexColor = vec4(0.1,0.8,0.2,1.0);
    }
}
